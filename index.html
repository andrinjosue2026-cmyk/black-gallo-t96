class InstaladorPWA {
  constructor(botonId) {
    this.boton = document.getElementById(botonId);
    this.eventoDiferido = null;
    this.configurarManifiesto();
    this.registrarServiceWorker();
    this.escucharEventosInstalacion();
  }

  async instalar() {
    if (!this.eventoDiferido) return false;
    this.eventoDiferido.prompt();
    const { outcome } = await this.eventoDiferido.userChoice;
    this.eventoDiferido = null;
    return outcome === 'accepted';
  }
}

class GestorAlmacenamiento {
  exportarJSON() {
    return JSON.stringify(this.cargar() || {}, null, 2);
  }
  importarJSON(textoJSON) {
    const parsed = JSON.parse(textoJSON);
    if (!Array.isArray(parsed.dias) || !Array.isArray(parsed.metas)) {
      throw new Error('Formato de respaldo inválido');
    }
    this.guardar(parsed);
    return parsed;
  }
}
